
board = [
    [5, 3, 0, 0, 7, 0, 0, 0, 0],
    [6, 0, 0, 1, 9, 5, 0, 0, 0],
    [0, 9, 8, 0, 0, 0, 0, 6, 0],
    [8, 0, 0, 0, 6, 0, 0, 0, 3],
    [4, 0, 0, 8, 0, 3, 0, 0, 1],
    [7, 0, 0, 0, 2, 0, 0, 0, 6],
    [0, 6, 0, 0, 0, 0, 2, 8, 0],
    [0, 0, 0, 4, 1, 9, 0, 0, 5],
    [0, 0, 0, 0, 8, 0, 0, 7, 9]
]

def possible(row, col, num):
    for i in range(9):
        if board[row][i] == num:
            return False
        if board[i][col] == num:
            return False

    start_row = (row // 3) * 3
    start_col = (col // 3) * 3

    for i in range(start_row, start_row + 3):
        for j in range(start_col, start_col + 3):
            if board[i][j] == num:
                return False

    return True

def find_empty():
    best_cell = None
    best_candidates = None

    for i in range(9):
        for j in range(9):
            if board[i][j] == 0:
                candidates = []

                for num in range(1, 10):
                    if possible(i, j, num):
                        candidates.append(num)

                if best_candidates is None or len(candidates) < len(best_candidates):
                    best_cell = (i, j)
                    best_candidates = candidates

                    if len(candidates) <= 1:
                        return best_cell, best_candidates

    return best_cell, best_candidates

def solve():
    cell, candidates = find_empty()

    if cell is None:
        return True

    if not candidates:
        return False

    row, col = cell

    for num in candidates:
        board[row][col] = num

        if solve():
            return True

        board[row][col] = 0

    return False

if solve():
    print("Sudoku Solved Successfully:")
    for row in board:
        print(row)
else:
    print("No solution exists")
