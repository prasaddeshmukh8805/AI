
location = input("Enter vacuum location (A/B): ")
status_A = input("Enter status of Room A (Clean/Dirty): ")
status_B = input("Enter status of Room B (Clean/Dirty): ")

if location == "A":
    if status_A == "Dirty":
        print("Room A is Dirty")
        print("Action: Suck")
    else:
        print("Room A is Clean")
        print("Action: Move Right")
else:
    if status_B == "Dirty":
        print("Room B is Dirty")
        print("Action: Suck")
    else:
        print("Room B is Clean")
        print("Action: Move Left")
